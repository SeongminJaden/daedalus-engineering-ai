# Optional node dependencies, installed separately from the core requirements.
#
# The analysis stack (physics, optimization, brain, surrogate) runs entirely
# in process and does not need these. They are required only to expose this
# engine as an MCP server, or to consume a capability from another node.
#
#   env -u PYTHONPATH .venv/bin/pip install -r requirements-nodes.txt
#
# The node layer imports this lazily and degrades to in-process routing when it
# is absent, so a user who never talks to another node does not install it.

mcp>=2.1

# CalculiX (ccx) is an EXTERNAL BINARY, not a Python package, and is optional.
# It provides an independently written FEA solver for cross-validating this
# project's own. Without it the CalculiX node reports itself unavailable and
# its tests skip; nothing else changes.
#
# Installed here WITHOUT sudo and without touching the system, by fetching the
# distribution packages and extracting them into a user prefix:
#
#   apt-get download calculix-ccx libspooles2.2
#   dpkg-deb -x calculix-ccx_*.deb  ~/.local/opt/calculix
#   dpkg-deb -x libspooles2.2_*.deb ~/.local/opt/calculix
#
# then a launcher on PATH that points at the extracted private libraries:
#
#   ~/.local/bin/ccx  ->  LD_LIBRARY_PATH=~/.local/opt/calculix/usr/lib/...
#                         exec ~/.local/opt/calculix/usr/bin/ccx "$@"
#
# `sudo apt install calculix-ccx` does the same thing system-wide if that is
# preferred. Verify with `ccx -v`.

# ---------------------------------------------------------------------------
# OpenFOAM, for the CFD cross-check (nodes/openfoam.py).
#
# Not a Python package. Installed the same sudo-free way as CalculiX:
#
#   apt-get download libopenfoam openfoam        # 64.4 MB total
#   dpkg-deb -x <each>.deb ~/.local/opt/openfoam # 415 MB unpacked
#
# The node sets WM_PROJECT_DIR, FOAM_ETC and LD_LIBRARY_PATH itself, so no
# shell wrapper or sourcing of etc/bashrc is needed. MPI comes from the system.
# Distribution is the ESI (openfoam.com) build, v1912 patch 200626.
# `sudo apt install openfoam` does the same thing system-wide.

# ---------------------------------------------------------------------------
# Pinocchio, for the multibody cross-check (nodes/pinocchio_node.py).
#
#   pip install pin
#
# 94 MB including its vendored boost, coal, assimp, octomap, qhull and
# urdfdom. It requires numpy, which this project already pins; check that a
# numpy upgrade is not pulled in, because Warp and torch are built against it.
pin

# ---------------------------------------------------------------------------
# Gmsh, for the meshing cross-check (nodes/gmsh_node.py).
#
#   pip install gmsh
#
# 39 MB, no dependencies. It bundles its own OpenCASCADE, which is what
# supplies the independent CAD volume. Note the narrow reach: this project's
# solver consumes only axis-aligned uniform hexahedra, so a Gmsh tetrahedral
# mesh cannot be run here at all.
gmsh

# ---------------------------------------------------------------------------
# MuJoCo, for general multibody contact simulation (nodes/mujoco_node.py).
#
#   pip install mujoco
#
# 19 MB plus six small pure-Python dependencies. Not robot specific: the
# capability registered is contact between rigid bodies in any mechanism.
# Note that MuJoCo has no coefficient of restitution and its contact is a
# regularised spring and damper, so bounce is measured from solref rather than
# set, and solref must be compatible with the timestep or energy inflates
# silently. See the node docstring for the measured numbers.
mujoco
